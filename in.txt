var line[]  resLines  = array.new_line()
var line[]  supLines  = array.new_line()
var label[] resLabels = array.new_label()
var label[] supLabels = array.new_label()

var float prevPH = na
var float prevPL = na
var string lastHighType = ""
var string lastLowType  = ""
var string trendState   = "Neutral"

ph = ta.pivothigh(high, pivotLen, pivotLen)
pl = ta.pivotlow(low, pivotLen, pivotLen)

nearestRes = array.size(resLines) > 0 ? line.get_y1(array.get(resLines, array.size(resLines)-1)) : na
nearestSup = array.size(supLines) > 0 ? line.get_y1(array.get(supLines, array.size(supLines)-1)) : na

var int   lastStructBar       = -1000
var float lastSweepHighLevel  = na
var float lastSweepLowLevel   = na

structCooldownOK = (bar_index - lastStructBar) > structCooldownBars

sweepHighRaw = showLiquidityEff and not na(nearestRes) and high > nearestRes and close < nearestRes
sweepLowRaw  = showLiquidityEff and not na(nearestSup) and low  < nearestSup and close > nearestSup

sweepHigh = sweepHighRaw and nearestRes != lastSweepHighLevel
sweepLow  = sweepLowRaw  and nearestSup != lastSweepLowLevel

if sweepHigh
    label.new(bar_index, high + atrVal*2.2, "SWEEP", style=label.style_label_down, color=color.new(color.purple,20), textcolor=color.white, size=size.tiny)
    lastSweepHighLevel := nearestRes
if sweepLow
    label.new(bar_index, low - atrVal*2.2, "SWEEP", style=label.style_label_up, color=color.new(color.purple,20), textcolor=color.white, size=size.tiny)
    lastSweepLowLevel := nearestSup

bosUpEvent   = not na(nearestRes) and ta.crossover(close, nearestRes)
bosDownEvent = not na(nearestSup) and ta.crossunder(close, nearestSup)

isChochUp = bosUpEvent and trendState == "Downtrend"
isBosUp   = bosUpEvent and trendState != "Downtrend"
isChochDn = bosDownEvent and trendState == "Uptrend"
isBosDn   = bosDownEvent and trendState != "Uptrend"

if showCHoCHEff and isChochUp and structCooldownOK
    label.new(bar_index, low - atrVal*1.2, "CHoCH", style=label.style_label_up, color=color.new(color.blue,10), textcolor=color.white, size=size.small)
    lastStructBar := bar_index
if showBOSEff and isBosUp and structCooldownOK
    label.new(bar_index, low - atrVal*1.2, "BOS", style=label.style_label_up, color=color.new(color.green,10), textcolor=color.white, size=size.small)
    lastStructBar := bar_index
if showCHoCHEff and isChochDn and structCooldownOK
    label.new(bar_index, high + atrVal*1.2, "CHoCH", style=label.style_label_down, color=color.new(color.blue,10), textcolor=color.white, size=size.small)
    lastStructBar := bar_index
if showBOSEff and isBosDn and structCooldownOK
    label.new(bar_index, high + atrVal*1.2, "BOS", style=label.style_label_down, color=color.new(color.red,10), textcolor=color.white, size=size.small)
    lastStructBar := bar_index

if bosUpEvent
    trendState := "Uptrend"
if bosDownEvent
    trendState := "Downtrend"

if showSR and not na(ph)
    highType = na(prevPH) ? "H" : (ph > prevPH ? "HH" : "LH")
    eqHigh = showLiquidityEff and not na(prevPH) and math.abs(ph - prevPH) < atrVal * eqTolerance
    lastHighType := highType
    prevPH := ph
    tagTxt = (showStructTags ? highType + "  " : "") + "Res " + str.tostring(ph, format.mintick) + (eqHigh ? "  EQH" : "")
    resCol = highType == "HH" ? color.orange : color.red
    ln = line.new(bar_index - pivotLen, ph, bar_index, ph, color=color.new(resCol,0), width=1, extend=extend.right)
    lb = label.new(bar_index - pivotLen, ph, tagTxt, style=label.style_label_down, color=color.new(resCol,80), textcolor=resCol, size=size.small)

// ░░░ 8. PROBABILITY ENGINE (clear, transparent, weighted formula) ░░░
// Every category (Trend / Momentum / Volume / Liquidity) is first turned
// into its OWN 0–100 sub-score (50 = neutral). The four sub-scores are then
// combined using adjustable weights (inputs 9b) that you set in %, so the
// final TOTAL is a straightforward weighted average — not a clipped sum.
// ══════════════════════════════════════════════════════════════
wSum = wTrendIn + wMomentumIn + wVolumeIn + wLiquidityIn
wT = wSum == 0 ? 0.25 : wTrendIn    / wSum
wM = wSum == 0 ? 0.25 : wMomentumIn / wSum
wV = wSum == 0 ? 0.25 : wVolumeIn   / wSum
wL = wSum == 0 ? 0.25 : wLiquidityIn/ wSum

// Trend sub-score: EMA stack + DI direction (only counted while ADX confirms a real trend) + structure
emaSub    = emaBullStack ? 100 : emaBearStack ? 0 : 50
diSub     = strongTrend ? (trendUpByDI ? 100 : 0) : 50
structSub = trendState == "Uptrend" ? 100 : trendState == "Downtrend" ? 0 : 50
trendPct  = (emaSub + diSub + structSub) / 3

// Momentum sub-score: RSI and %K are already native 0-100 oscillators, so just average them
momentumPct = math.max(0, math.min(100, (rsiVal + kLine) / 2))

// Volume sub-score: money flow direction + confirming volume spike
cmfSub     = cmfBullish ? 100 : 0
volSpikeSub= highVolume ? (close > open ? 100 : 0) : 50
volumePct  = (cmfSub + volSpikeSub) / 2

// Liquidity sub-score: most recent sweep direction + BOS/CHoCH direction
sweepSub = sweepLow ? 100 : sweepHigh ? 0 : 50
bosSub   = (isBosUp or isChochUp) ? 100 : (isBosDn or isChochDn) ? 0 : 50
liquidityPct = (sweepSub + bosSub) / 2

totalPct = trendPct*wT + momentumPct*wM + volumePct*wV + liquidityPct*wL
bullBias = totalPct >= 50

// AI Confidence: distance of TOTAL from the neutral 50% line, rescaled to 0-100
confidencePct   = math.abs(totalPct - 50) * 2
confidenceLabel = confidencePct >= 70 ? "HIGH" : confidencePct >= 40 ? "MEDIUM" : "LOW"
confidenceColor = confidencePct >= 70 ? color.lime : confidencePct >= 40 ? color.orange : color.gray

// ░░░ 17. GODZILLA LEVERAGE MODULE ░░░
// Direction (I_t) = proxy order-flow imbalance using up-volume vs down-volume
// (real bid/ask book volume is not available inside Pine Script).
// Sigma (σ_t) = smoothed ATR-as-%-of-price, standing in for GARCH forecasted
// volatility, since Pine has no native GARCH/ARCH engine.
// Max Leverage_t = 1 / (1 + σ_t ^ L0), then scaled into an actual x-leverage
// number by your Absolute Max Leverage input.
// ══════════════════════════════════════════════════════════════
sumUpVol   = math.sum(close > close[1] ? volume : 0.0, imbalanceLen)
sumDnVol   = math.sum(close < close[1] ? volume : 0.0, imbalanceLen)
totalFlow  = sumUpVol + sumDnVol
imbalance_It = totalFlow == 0 ? 0.0 : (sumUpVol - sumDnVol) / totalFlow   // -1..+1, proxy for I_t

godzillaDir      = imbalance_It > 0 ? "LONG" : imbalance_It < 0 ? "SHORT" : "NEUTRAL"
godzillaDirColor = imbalance_It > 0 ? color.lime : imbalance_It < 0 ? color.red : color.gray

sigmaPctRaw  = close == 0 ? 0.0 : (atrVal / close) * 100         // ATR as % of price
sigmaSmooth  = ta.sma(sigmaPctRaw, sigmaLen)                     // smoothed, proxy for forecasted variance/vol
sigmaAdj     = (sigmaSmooth / 100) * sigmaScale                  // scaled decimal input to the leverage formula

leverageFactor  = 1 / (1 + math.pow(sigmaAdj, godzillaL0))       // 0..1 factor from the Max Leverage formula
suggestedLev    = leverageFactor * leverageCap                    // scaled into actual x-leverage
suggestedLevTxt = str.tostring(suggestedLev, "#.##") + "x"
